+// SPDX-License-Identifier: AGPL-3.0-only
+//! Optional fee-only deployment contract. Existing signed artifacts can be
+//! submitted unchanged; fresh fee signatures require an authenticated caller.
+use crate::{
+    domain::transaction::evm::PriceParams,
+    models::{EvmTransactionData, EvmTransactionRequest, TransactionError},
+};
+pub fn enabled() -> bool {
+    std::env::var("WPK_FEE_ONLY_REPLACEMENTS").as_deref() == Ok("1")
+}
+pub fn prices(data: &EvmTransactionData, prices: &PriceParams) -> Result<(), TransactionError> {
+    if data.gas_limit.is_none()
+        || data.gas_price.is_some()
+        || data.max_fee_per_gas.is_none()
+        || data.max_priority_fee_per_gas.is_none()
+        || prices.gas_price.is_some()
+        || prices.max_fee_per_gas != data.max_fee_per_gas
+        || prices.max_priority_fee_per_gas != data.max_priority_fee_per_gas
+    {
+        return Err(TransactionError::ValidationError(
+            "fee-only contract requires exact EIP-1559 prices and gas limit".into(),
+        ));
+    }
+    Ok(())
+}
+pub fn same_call(
+    old: &EvmTransactionData,
+    request: &EvmTransactionRequest,
+) -> Result<(), TransactionError> {
+    if old.to != request.to
+        || old.value != request.value
+        || old.data != request.data
+        || old.gas_limit != request.gas_limit
+        || old.nonce.is_none()
+        || request.gas_price.is_some()
+        || request.max_fee_per_gas.is_none()
+        || request.max_priority_fee_per_gas.is_none()
+    {
+        return Err(TransactionError::ValidationError(
+            "replacement must retain the original call and nonce with explicit EIP-1559 prices"
+                .into(),
+        ));
+    }
+    Ok(())
+}
